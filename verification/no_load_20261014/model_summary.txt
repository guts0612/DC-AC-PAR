SCRIPT function out_A  = fcn(err_A, Kp, Kr, Wc, Ts)

W0 = 50*2*pi;


persistent err_A1;
persistent err_A2;
persistent out_A1;
persistent out_A2;


%%对三相静态变量初始化
if isempty(err_A1)
    err_A1 = 0;
end
if isempty(err_A2)
    err_A2 = 0;
end
if isempty(out_A1)
    out_A1 = 0;
end
if isempty(out_A2)
    out_A2 = 0;
end


%%Discrete
t1 = Wc * Ts;
t2 = W0^2 * Ts^2;
t3 = t2 + 4*t1 + 4;

b0 = (Kp*t2 + 4*(Kp+Kr)*t1 + 4*Kp) / t3;
b1 = (2*Kp*t2 - 8*Kp) / t3;
b2 = (Kp*t2 + 4*Kp - 4*(Kp+Kr)*t1) / t3;
a1 = (2*t2 - 8) / t3;
a2 = (t2 - 4*t1 + 4) / t3;
%%限制输出
if b0>3000
    b0=0;
end
if b0<-3000
    b0=0;
end

%%计算输出量
out_A = b0*err_A + b1*err_A1 + b2*err_A2 - a1*out_A1 - a2*out_A2;
out_A2 = out_A1;
out_A1 = out_A;
err_A2 = err_A1;
err_A1 = err_A;
